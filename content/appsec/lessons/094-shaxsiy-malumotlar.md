      passport: string;

    const ALLOWED: Record<string, (keyof Row)[]> = {
      report: ["id", "name", "trips"],
      support: ["id", "name", "phone"],
      billing: ["id", "name", "plan"],

    function project(row: Row, purpose: string): Record<string, unknown> {
      const fields = Object.hasOwn(ALLOWED, purpose) ? ALLOWED[purpose] : [];
      const out: Record<string, unknown> = {};
      for (const f of fields) if (row[f] !== undefined) out[f] = row[f];
      return out;

## Muammo

Platformada haydovchining ismi, telefon raqami, guvohnomasi, GPS izi, video yozuv va yoqilg'i hisoboti saqlanadi. Bular ustunlar, qonun uchun esa shaxsga doir ma'lumotlar bo'lishi mumkin. Dasturchi amalda shu vaziyatlarga tushadi.

Birinchisi: arzon bulut kerak va u chet elda. Haydovchilar yuzi DMS kamerada tanilsa va bu yozuvlar o'sha bulutga ketsa, 27-1-modda bo'yicha muammo bo'lishi mumkin: ba'zi toifalar O'zbekistonda saqlanishi shart.

Ikkinchisi: debug uchun logga butun so'rov tanasi yoziladi, unda telefon raqami va pasport bor. Log hamma dasturchiga ochiq, hech qachon o'chirilmaydi.

Uchinchisi: "keyin kerak bo'lar" deb hamma yozuv abadiy saqlanadi, baza esa reyestrda yo'q, subyekt so'rov yuborsa ma'lumotni topadigan va o'chiradigan funksiya ham yo'q.

Har biri texnik nuqson (1-dars) va huquqiy nuqson: Ma'muriy va Jinoyat kodekslari bo'yicha javobgarlikka olib kelishi mumkin. Matnlar 2026-11-11 da lex.uz'dan o'qilgan; qonun 2026-yilda uch marta o'zgargan, shuning uchun havola va o'qilgan sanani yozing.

## Asosiy g'oya

O'RQ-547 (02.07.2019) 36 moddadan iborat, 01.10.2019 da kuchga kirgan. Dasturchi uchun u sakkiz qoidaga tushadi: qoida, nega (modda), amalda nima qilasiz.

**1. Haydovchiga bog'langan har yozuvni shaxsga doir ma'lumot deb hisoblang.** Nega: 4-moddaga ko'ra shaxsga doir ma'lumot - muayyan jismoniy shaxsga taalluqli yoki uni identifikatsiya qilish imkonini beradigan axborot; "ishlov berish" ta'rifi log yozish, hisobotga qo'shish, boshqa servisga yuborish va o'chirishni ham qamraydi. GPS izi identifikatsiya imkonini bersa, ta'rifga tushadi (kursning talqini). Amalda: har maydon uchun toifa belgilang (data inventory).

**2. Ishlovdan oldin asos va maqsadni yozing; maqsad o'zgarsa yangi rozilik oling.** Nega: 18-modda asoslarni sanaydi, rozilik yagona asos emas; 19-modda maqsad o'zgarsa yangi rozilikni talab qiladi; 21 va 31-moddalar rozilik olinganligi faktini isbotlashni talab qiladi. Amalda: kim, qachon, matnning qaysi versiyasi va qaysi maqsad uchun rozilik bergani va qachon qaytarib olgani saqlanadi.

**3. Har so'rovda faqat zarur maydonni qaytaring.** Nega: 10 va 19-moddalar "zarur va yetarli" hajmni, 12-modda faqat bildirilgan maqsad uchun foydalanishni talab qiladi. Amalda: maqsad bo'yicha default-deny proyeksiya (Misol bo'limi).

**4. Muddat tugagach yo'q qiling yoki egasizlantiring.** Nega: 10-modda saqlash muddatini maqsadga erishilgan sana bilan belgilaydi; 16 va 17-moddalar egasizlantirish va yo'q qilishni ta'riflaydi. Amalda: retention job, zaxira nusxa, kesh va logdagi nusxalar bilan birga. Ismni almashtirib ID qoldirish egasizlantirish emas.

**5. Subyekt so'rovlarini oldindan quring.** Nega: 11, 22, 23, 24 va 30-moddalar subyektga o'zgartirish, axborot olish, xabardor qilinish, avtomatlashtirilgan qarorga e'tiroz va ishlovni to'xtatishni talab qilish huquqini beradi, muddatlari 3 va 10 kun. Amalda: topish, eksport, tuzatish, to'xtatish, o'chirish va rozilikni qaytarish funksiyalari; driver scoring uchun e'tiroz va qo'lda ko'rib chiqish yo'li.

**6. Saqlash joyini ma'lumot toifasi bo'yicha tanlang.** Nega: 27-1-modda biometrik va genetik ma'lumotni hamda telekommunikatsiya operatorlaridagi foydalanuvchi ma'lumotini O'zbekistonda saqlashni shart qiladi; qolganlari uchun uch shartdan biri kerak; 15-modda transchegaraviy uzatishni tartibga soladi. Ijro: birinchi shart bo'yicha ro'yxat VMQ-415 bilan tasdiqlangan; qolgan ikki shart bo'yicha hujjat topilmadi. Amalda: toifa bo'yicha ruxsat etilgan joylar konfiguratsiyada va deploy vaqtida tekshiriladi.

**7. Biometrik va maxsus ma'lumotni alohida rejimda yuriting.** Nega: 25-modda maxsus ma'lumotga ishlov berishni, istisnolardan tashqari, taqiqlaydi; 26-modda biometrik va genetik ma'lumotga faqat roziligi bilan (istisnolardan tashqari) ruxsat beradi. Yuz bilan identifikatsiya shu toifaga tushishi mumkin (kursning talqini). Amalda: DMS kamera ma'lumotini alohida toifa va alohida saqlash joyi bilan yuriting.

**8. Himoya, maxfiylik va hodisa jarayonini quring.** Nega: 27 va 28-moddalar himoya choralari va maxfiylikni, 31-modda huquqiy, tashkiliy va texnik himoyani talab qiladi; hodisada O'RQ-764 16-moddasi ishlaydi. Amalda: access logging, hodisa jarayoni (93-dars), reyestr arizasi (20-modda, VMQ-71), himoya darajasi (VMQ-570).

## Chuqurroq: privacy by design kodda

Qoidalarning muhandislik oqibatlari:

- Data inventory. Har maydon uchun toifa, maqsad, huquqiy asos, saqlash muddati, joyi va qabul qiluvchilar yoziladi: 22-modda aynan shuni subyektga berishni talab qiladi. Tasniflanmagan maydon CI'da to'xtatiladi.
- Minimizatsiya. Default-deny proyeksiya 10 va 19-moddalardagi "zarur va yetarli" talabning kod ifodasi.
- Access logging. Xodim faqat o'z majburiyati doirasida foydalanadi (12), maxfiylik saqlanadi (28): kim, qachon, qaysi maqsadda kirgani logga yoziladi; logda qiymat emas ID bo'ladi, logning ham muddati bor (79 va 93-darslar). 16-modda hodisa uchun raqamli izlarni saqlashni talab qiladi.
- Hosting joyi. Biometrik va genetik ma'lumot faqat O'zbekistonda (27-1).
- User Privacy Protection Cheat Sheet qo'shadi: uzatishda va saqlashda kriptografiya, remote session invalidation, tashqi kontentdan IP sizishining oldini olish.

### 3. Tuzatilgan kod

```ts run
type Driver = { id: string; name: string; phone: string; passport: string; address: string; trips: number; lastSeen: string };
const drivers: Driver[] = [
  { id: "d1", name: "Aziz K.", phone: "+998900000001", passport: "AA0000001", address: "Toshkent, namunaviy 1", trips: 42, lastSeen: "2026-01-10" },
  { id: "d2", name: "Bahrom T.", phone: "+998900000002", passport: "AA0000002", address: "Toshkent, namunaviy 2", trips: 17, lastSeen: "2025-02-01" },
];

// 1) Maqsad bo'yicha proyeksiya (default-deny)
const PURPOSES: Record<string, (keyof Driver)[]> = {
  report: ["id", "name", "trips"],
  support: ["id", "name", "phone"],
};
function project(d: Driver, purpose: string): Partial<Driver> {
  const out: Partial<Driver> = {};
  for (const f of Object.hasOwn(PURPOSES, purpose) ? PURPOSES[purpose] : []) (out as Record<string, unknown>)[f] = d[f];
  return out;
}

// 2) Saqlash muddati: muddati o'tganlar egasizlantiriladi (shaxsni bog'lovchi maydonlar olib tashlanadi)
const RETENTION_DAYS = 365;
function anonymize(list: Driver[], today: string) {
  const limit = Date.parse(today) - RETENTION_DAYS * 86400000;
  return list.map((d) =>
    Date.parse(d.lastSeen) < limit ? { trips: d.trips, region: "Toshkent" } : d,
  );
}

// 3) Subyekt so'rovi: bitta odam haqida hamma narsa
const exportFor = (id: string) => drivers.filter((d) => d.id === id);

console.log("report:", JSON.stringify(drivers.map((d) => project(d, "report"))));
console.log("noma'lum maqsad:", JSON.stringify(project(drivers[0], "constructor")));
console.log("anonimlashtirish:", JSON.stringify(anonymize(drivers, "2026-11-11")));
console.log("subyekt eksporti maydonlari:", Object.keys(exportFor("d1")[0]).join(","));
```

### Saqlash joyi kodda

```ts run
type Category = "oddiy" | "biometrik" | "genetik" | "telekom";
type Place = { country: string; basis?: "standart-shartnoma" | "tan-olingan-davlat" | "xalqaro-standart" };

// Siyosat (jamoa belgilaydi): bu kod faqat siyosatni qo'llaydi
const MUST_STAY = new Set<Category>(["biometrik", "genetik", "telekom"]);
function storageAllowed(cat: Category, place: Place): { ok: boolean; why: string } {
  if (place.country === "UZ") return { ok: true, why: "O'zbekiston hududida" };
  if (MUST_STAY.has(cat)) return { ok: false, why: "bu toifa O'zbekistonda saqlanishi shart" };
  if (!place.basis) return { ok: false, why: "chet elda saqlash uchun asos ko'rsatilmagan" };
  return { ok: true, why: `asos: ${place.basis}` };
}
```

## Stack'da: Prisma, PostgreSQL va NestJS

Maydon tasnifi uchun `column_classification` jadvali, CI testi tasniflanmagan ustunni topadi. Prisma'da proyeksiya `select` bilan so'rov darajasida (ma'lumot xotiraga ham kelmaydi).
